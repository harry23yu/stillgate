# Settings (app preferences) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Settings screen for app preferences (text size, reduced motion, and — Phase 2 — dim/dark mode), reached from the Home gear, distinct from the sensory Profile.

**Architecture:** A `settings.ts` AsyncStorage store feeds a `SettingsProvider`/`useSettings()` React context mounted at the root layout so preferences apply live. Phase 1 wires text size through `ThemedText` and reduced motion through animation sites. Phase 2 makes `useTheme()` scheme-aware and migrates static `StyleSheet.create(Colors.x)` blocks to a `useThemedStyles()` helper so a dark palette repaints the whole app.

**Tech Stack:** React Native 0.88 + Expo SDK 58, expo-router, TypeScript, AsyncStorage, react-native-reanimated (`useReducedMotion`), `useColorScheme`.

Spec: `docs/superpowers/specs/2026-10-24-settings-design.md`.

## Global Constraints

- **Expo Go only — no native modules.** No new runtime dependencies (all needed APIs already ship: AsyncStorage, reanimated, `useColorScheme`, expo-status-bar, expo-system-ui).
- **No secrets** in code or commits.
- **Commit messages:** do NOT add a `Co-Authored-By` trailer (project preference).
- **Accessibility is a requirement:** every control has a screen-reader label and a ≥ 44 px target; selected state conveyed via text + `accessibilityState`, not color alone; dark palette (Phase 2) ≥ 4.5:1 body-text contrast.
- **Design tokens** come from `@/constants/theme`. Reuse `ThemedText`, `ThemedView`, `Chip`, `Button`, `Card`. Do not hardcode hex outside `theme.ts`.
- **Path alias:** `@/` maps to `src/`.
- **Verification gate for UI:** `npx tsc --noEmit` + `npx expo lint` + manual on a real iPhone. (A jest data-layer harness may exist from the layover plan; if so, the store task adds a unit test, otherwise typecheck + device covers it.)

- [ ] **Step 1: Create the store**

Create `src/data/settings.ts`:

```ts
import AsyncStorage from '@react-native-async-storage/async-storage';

const STORAGE_KEY = 'stillgate.settings';

export type ThemePreference = 'system' | 'light' | 'dark';

export interface AppSettings {
  /** Phase 2 acts on this; Phase 1 leaves it at 'system'. */
  theme: ThemePreference;
  /** Font-size multiplier: 1 (default) | 1.15 | 1.3. */
  textScale: number;
  reducedMotion: boolean;
}

export const defaultAppSettings: AppSettings = {
  theme: 'system',
  textScale: 1,
  reducedMotion: false,
};

export async function getSettings(): Promise<AppSettings> {
  const raw = await AsyncStorage.getItem(STORAGE_KEY);
  if (!raw) return defaultAppSettings;
  try {
    return { ...defaultAppSettings, ...JSON.parse(raw) };
  } catch {
    return defaultAppSettings;
  }
}

export async function saveSettings(settings: AppSettings): Promise<void> {
  await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
}
```

- [ ] **Step 1: Create the provider**

Create `src/contexts/settings.tsx`:

```tsx
import { createContext, useCallback, useContext, useEffect, useState, type ReactNode } from 'react';
import { type AppSettings, defaultAppSettings, getSettings, saveSettings } from '@/data/settings';

interface SettingsContextValue {
  settings: AppSettings;
  updateSettings: (patch: Partial<AppSettings>) => void;
  loaded: boolean;
}

const SettingsContext = createContext<SettingsContextValue>({
  settings: defaultAppSettings,
  updateSettings: () => {},
  loaded: false,
});

export function SettingsProvider({ children }: { children: ReactNode }) {
  const [settings, setSettings] = useState<AppSettings>(defaultAppSettings);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    let active = true;
    getSettings().then((stored) => {
      if (!active) return;
      setSettings(stored);
      setLoaded(true);
    });
    return () => {
      active = false;
    };
  }, []);

  const updateSettings = (patch: Partial<AppSettings>) => {
    setSettings((current) => {
      const next = { ...current, ...patch };
      void saveSettings(next);
      return next;
    });
  };

  return (
    <SettingsContext.Provider value={{ settings, updateSettings, loaded }}>
      {children}
    </SettingsContext.Provider>
  );
}

export function useSettings(): SettingsContextValue {
  return useContext(SettingsContext);
}
```

- [ ] **Step 1: Multiply font size and line height by the scale**

In `src/components/themed-text.tsx`, add the settings hook and apply the multiplier. The `styles[type]` objects expose numeric `fontSize`/`lineHeight`:

```tsx
import { StyleSheet, Text, type TextProps } from 'react-native';

import { Fonts, ThemeColor } from '@/constants/theme';
import { useSettings } from '@/contexts/settings';
import { useTheme } from '@/hooks/use-theme';

export type ThemedTextProps = TextProps & {
  type?: 'default' | 'title' | 'subtitle' | 'eyebrow' | 'small' | 'smallBold';
  themeColor?: ThemeColor;
};

export function ThemedText({ style, type = 'default', themeColor, ...rest }: ThemedTextProps) {
  const theme = useTheme();
  const { settings } = useSettings();
  const base = styles[type];
  const scaled = {
    fontSize: base.fontSize * settings.textScale,
    lineHeight: base.lineHeight * settings.textScale,
  };

  return <Text style={[{ color: theme[themeColor ?? 'text'] }, base, scaled, style]} {...rest} />;
}
```

(The rest of the file — the `styles` StyleSheet — is unchanged.)

Replace `src/hooks/use-theme.ts`:

```ts
import { useColorScheme } from 'react-native';

import { Colors, ColorsDark, type Palette } from '@/constants/theme';
import { useSettings } from '@/contexts/settings';

/** Returns the active palette based on the theme preference and the OS scheme. */
export function useTheme(): Palette {
  const { settings } = useSettings();
  const system = useColorScheme();
  const scheme = settings.theme === 'system' ? system ?? 'light' : settings.theme;
  return scheme === 'dark' ? ColorsDark : Colors;
}
```

- [ ] **Step 3: Create the themed-styles helper**

Create `src/hooks/use-themed-styles.ts`:

```ts
import { useMemo } from 'react';
import { StyleSheet } from 'react-native';

import type { Palette } from '@/constants/theme';
import { useTheme } from '@/hooks/use-theme';

/**
 * Builds a StyleSheet from the active palette and rebuilds it when the palette
 * changes. The factory must depend only on the palette (not on props/state).
 */
export function useThemedStyles<T extends StyleSheet.NamedStyles<T>>(factory: (c: Palette) => T): T {
  const theme = useTheme();
  return useMemo(() => StyleSheet.create(factory(theme)), [theme]);
}
```

import { Pressable } from 'react-native';

import { useThemedStyles } from '@/hooks/use-themed-styles';

type ChipProps = {
  label: string;
  selected: boolean;
  onPress: () => void;
  accessibilityHint?: string;
  accessibilityLabel?: string;
};

export function Chip({ label, selected, onPress, accessibilityHint, accessibilityLabel }: ChipProps) {

      onPress={onPress}
      accessibilityRole="button"
      accessibilityState={{ selected }}
      accessibilityLabel={accessibilityLabel ?? label}
      accessibilityHint={accessibilityHint}
      hitSlop={4}
      style={[styles.chip, selected ? styles.chipSelected : styles.chipUnselected]}>

}

import { DefaultTheme, Stack, ThemeProvider } from 'expo-router';
import { StatusBar } from 'expo-status-bar';
import { useFonts } from 'expo-font';
import * as SplashScreen from 'expo-splash-screen';
import { useColorScheme } from 'react-native';

import { LaunchScreen } from '@/components/launch-screen';
import { SettingsProvider, useSettings } from '@/contexts/settings';
import { useTheme } from '@/hooks/use-theme';

SplashScreen.preventAutoHideAsync();

function ThemedApp() {
  const palette = useTheme();
  const { settings } = useSettings();
  const system = useColorScheme();
  const scheme = settings.theme === 'system' ? system ?? 'light' : settings.theme;

  const navTheme = {
    ...DefaultTheme,
    dark: scheme === 'dark',
    colors: {
      ...DefaultTheme.colors,
      primary: palette.primary,
      background: palette.background,
      card: palette.surface,
      text: palette.text,
      border: palette.border,
    },
  };

  return (
    <ThemeProvider value={navTheme}>
      <StatusBar style={scheme === 'dark' ? 'light' : 'dark'} />
      <LaunchScreen />
      <Stack screenOptions={{ headerShown: false }}>
        <Stack.Screen name="(tabs)" />
        <Stack.Screen name="gate-change" options={{ presentation: 'card' }} />
        <Stack.Screen name="flight-lookup" options={{ presentation: 'modal' }} />
        <Stack.Screen name="settings" options={{ presentation: 'modal' }} />
      </Stack>
    </ThemeProvider>
  );
}

export default function RootLayout() {
  const [fontsLoaded] = useFonts({
    'AtkinsonHyperlegible-Regular': require('@/assets/fonts/AtkinsonHyperlegible-Regular.ttf'),
    'AtkinsonHyperlegible-Bold': require('@/assets/fonts/AtkinsonHyperlegible-Bold.ttf'),
  });

  if (!fontsLoaded) return null;

  return (
    <SettingsProvider>
      <ThemedApp />
    </SettingsProvider>
  );
}
